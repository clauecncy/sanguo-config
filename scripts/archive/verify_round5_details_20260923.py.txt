"""Verify two full level-ten screenshots without overwriting lower levels."""
from contextlib import closing
from datetime import datetime
import json
import shutil
import sqlite3
from project_data import ROOT, atomic_json
from verify_event_round3_details_20260923 import add_source, add_observation

DETAILS = [
    ("指点乾坤", "主动", "50%", "令我军武力最高单体对敌军全体造成80%兵刃伤害，令我军智力最高单体对敌军全体造成80%谋略伤害。", "433a55e4-9787-442d-b547-cdf917376808"),
    ("以静制动", "被动", "100%", "奇数回合，自身无法进行普通攻击，自身和随机友军单体受到兵刃伤害、普通攻击伤害、追击战法伤害减少20%（受统率影响）。偶数回合，自身无法发动主动战法，自身和随机友军单体受到谋略伤害、主动战法伤害减少20%（受统率影响）。", "ab93e2b4-e51c-452d-8ae4-8068e95f859a"),
]

def main():
    event = ROOT / "user/bixianjue/演武/s2-2026-09-23"
    files = [ROOT / f for f in ("game/game.sqlite3", "game/sources/public-baseline.sql", "game/sources/tactic_level_observations.json", "game/sources/evidence-index.json")]
    backup = ROOT / "notes/backups" / (datetime.now().strftime("%Y-%m-%d-%H%M%S") + "-round5-details")
    for f in files + [event / "event.json", event / "第5轮选战法.json", ROOT / "notes/业务逻辑.md", ROOT / "notes/变更记录.md"]:
        dest = backup / f.relative_to(ROOT)
        dest.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(f, dest)
    obs = json.loads(files[2].read_text(encoding="utf-8"))
    index = json.loads(files[3].read_text(encoding="utf-8"))
    with closing(sqlite3.connect(files[0])) as live, closing(sqlite3.connect(":memory:")) as c:
        assert files[1].read_text(encoding="utf-8") == "\n".join(live.iterdump()) + "\n"
        live.backup(c)
        for name, kind, rate, effect, image_id in DETAILS:
            filename = f"2026-09-23-{name}10级.png"
            target = ROOT / "game/sources/evidence" / filename
            if target.exists():
                raise ValueError("Evidence already recorded")
            shutil.copy2(f"C:/Users/BIXIAN~1/AppData/Local/Temp/codex-clipboard-{image_id}.png", target)
            tid, quality = c.execute("select id,quality from tactics where name=?", (name,)).fetchone()
            sid, path = add_source(c,index,name+"10级",filename,"用户第5轮完整10级详情截图；不外推其他等级红度")
            c.execute("update tactics set description_raw=?,description_level=10,tactic_type=?,activation_rate=?,source_id=?,trust_status='可信',trust_reason='用户本期完整10级截图逐字段核实',verification_status='Steam已核',screenshot_verified_fields=?,updated_at='2026-09-23' where id=?", (effect,kind,rate,sid,json.dumps(["name","tactic_type","activation_rate","description_raw","description_level"],ensure_ascii=False),tid))
            add_observation(c,obs,tid,name,quality,kind,rate,effect,sid,path,"bixianjue第5轮完整10级详情；红度未单独核定")
            from PIL import Image
            Image.open(target).verify()
        c.commit()
        sql = "\n".join(c.iterdump()) + "\n"
        with closing(sqlite3.connect(":memory:")) as test:
            test.executescript(sql)
            assert list(test.iterdump()) == list(c.iterdump())
        c.backup(live)
    files[1].write_text(sql,encoding="utf-8")
    atomic_json(files[2],obs)
    atomic_json(files[3],index)
    p = event / "第5轮选战法.json"
    d = json.loads(p.read_text(encoding="utf-8"))
    d.update(recommended_group=2,status="建议第二组，尚未实际选择",missing_details=[],evidence_note="指点乾坤及以静制动已按本期完整10级图核实；烈火旧满级红度未知，不将其数值硬套演武白板。")
    atomic_json(p,d)
    p = event / "event.json"
    d = json.loads(p.read_text(encoding="utf-8"))
    d["stage"] = "前4轮已确认，10将14战法；第5轮建议第二组，未实际选择"
    if "第5轮选战法建议.md" not in d["files"]:
        d["files"].append("第5轮选战法建议.md")
    atomic_json(p,d)
    print("Two screenshots verified; independent SQL rebuild equal; round5 recommendation=2, confirmed=null.")
    print(backup)

if __name__ == "__main__":
    main()
